# Zone Two Health and Performance

# Retrieval and search crawlers are deliberately ALLOWED. Blocking these removes
# the clinic from AI answers and search results, which is the opposite of what a
# local clinic wants. Training-only crawlers (GPTBot, ClaudeBot,
# Google-Extended, Applebot-Extended) are also allowed by default; blocking them
# would not affect discoverability either way.
User-agent: *
Allow: /

# No indexable content, and no reason to crawl.
Disallow: /api/

# /book/, /concepts/ and /account/ are intentionally NOT disallowed: every page
# there carries a noindex meta, and a blocked URL is never fetched, so the
# crawler would never see the noindex and could still list the bare URL.

Sitemap: https://zone2hp.com/sitemap-index.xml
